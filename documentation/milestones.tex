\begin{abstract}
This document defines multi-phase milestones for the Vehicle Dynamics backend.
Each milestone follows the Phase~1 pattern: directory and documentation updates,
then a concrete runnable entry-point change. Frontend work remains deferred
until after the GPU DEM path is established.
\end{abstract}

\begin{IEEEkeywords}
milestones, vehicle dynamics, DEM, project plan, verification
\end{IEEEkeywords}

\section{Milestone Pattern}
Every milestone delivers three artifact classes, matching Phase~1:
\begin{enumerate}
  \item \textbf{Structure} --- directories and/or modules.
  \item \textbf{Documentation} --- IEEE \LaTeX\ updates under
        \texttt{documentation/}.
  \item \textbf{Execution} --- \texttt{main.py} (or package entry) runs a
        visible, testable behavior for that phase.
\end{enumerate}

\section{M1 --- Scaffold (Complete)}
\begin{enumerate}
  \item Create \texttt{documentation/}, \texttt{src/}, \texttt{tests/}.
  \item Add IEEE SRS, SDD, testing \LaTeX; basic README; CI YAML.
  \item \texttt{main.py} with \texttt{if \_\_name\_\_ == \"\_\_main\_\_\"} smoke print.
\end{enumerate}

\section{M2 --- Backend Package}
\begin{enumerate}
  \item Create \texttt{src/vehicle\_dynamics/} with \texttt{\_\_init\_\_.py},
        \texttt{config.py}, package \texttt{\_\_main\_\_.py}.
  \item Update SDD module map; amend SRS FR-2 config notes.
  \item \texttt{main.py} imports package and prints loaded default config.
\end{enumerate}
\textbf{Exit criteria:} \texttt{python main.py} and
\texttt{python -m vehicle\_dynamics} (with \texttt{PYTHONPATH=src}) succeed;
pytest green.

\section{M3 --- Vehicle Core}
\begin{enumerate}
  \item Add \texttt{vehicle/}, \texttt{integrate/}, \texttt{scenarios/},
        \texttt{io/} modules (dual-track 3-DOF + tire stubs).
  \item Expand SRS vehicle requirements; testing doc gains vehicle TCs.
  \item \texttt{main.py} runs a short step-steer and writes CSV under
        \texttt{out/}.
\end{enumerate}
\textbf{Exit criteria:} CSV exists; unit tests cover integrator smoke.

\section{M4 --- DEM Soil (Small $N$)}
\begin{enumerate}
  \item Add \texttt{dem/} particle buffers and CPU contact/step stubs.
  \item Document DEM parameters in SRS/SDD; testing doc DEM unit cases.
  \item \texttt{main.py} advances a tiny particle bed (hundreds of bodies)
        and logs summary stats.
\end{enumerate}
\textbf{Exit criteria:} Deterministic tiny DEM test passes on CI (CPU).

\section{M5 --- Tire--Soil Coupling}
\begin{enumerate}
  \item Add \texttt{coupling/} force exchange between wheels and particles.
  \item Update SDD data-flow diagrams; testing doc integration cases.
  \item \texttt{main.py} runs coupled short maneuver on small DEM field.
\end{enumerate}
\textbf{Exit criteria:} Integration test asserts finite forces and stable step.

\section{M6 --- GPU Scale Path}
\begin{enumerate}
  \item Implement GPU DEM kernels (default: NVIDIA Warp) behind \texttt{dem/}.
  \item Add performance notes to testing doc; NFR scale checklist in SRS.
  \item \texttt{main.py} flag/config selects CPU-tiny vs GPU-large modes.
\end{enumerate}
\textbf{Exit criteria:} Documented large-$N$ run on GPU hardware; CI keeps
CPU-tiny path green without GPU.

\section{M7 --- Front End (Deferred)}
Visualization or interactive client consuming backend outputs. Not scheduled
until M6 exit criteria are met.
